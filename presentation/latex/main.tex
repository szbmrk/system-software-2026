\section{Garbage Collection Background}

\begin{frame}{Why Garbage Collection Matters}
	\textbf{Managed languages} (Java, C\#) allocate many \textbf{small, short-lived} objects, so the garbage collector \textbf{directly determines} performance.

	\medskip
	Applications make conflicting demands:

	\begin{itemize}
		\item real-time: \textbf{pause-time guarantees}
		\item embedded: \textbf{space efficiency}
		\item servers: \textbf{throughput}
	\end{itemize}
\end{frame}

\begin{frame}{Tracing Garbage Collection}
	\begin{columns}[c]
		\begin{column}{0.52\textwidth}
			A tracing collector has three jobs:

			\begin{itemize}
				\item \textbf{Allocation} of new objects.
				\item \textbf{Identification}: mark every object reachable from the roots.
				\item \textbf{Reclamation} of unmarked objects.
			\end{itemize}

			\medskip
			An object is \texttt{dead} if we can't reach it from roots.
		\end{column}

\begin{frame}{The Canonical Tracing Collectors}
	The paper compares immix against the three canonical tracing collectors:

	\begin{itemize}
		\item \textbf{Mark-sweep (MS):} non-moving, objects stay where they were allocated.
		\item \textbf{Semi-space (SS):} moving, copies live objects into a second space.
		\item \textbf{Mark-compact (MC):} moving, slides live objects together within the same space.
	\end{itemize}

	\begin{exampleblock}{Building blocks}
		\vspace{0.1em}
		More sophisticated collectors, such as generational ones, are built by combining these.
	\end{exampleblock}
\end{frame}

	\end{itemize}

\begin{frame}{The Space-Time Trade-off}
	\begin{columns}[c]
		\begin{column}{0.54\textwidth}
			MS, SS and MC from 1$\times$ to 6$\times$ the minimum heap:

			\begin{itemize}
				\item \textbf{Small heap:} GC runs often, fast collection wins (\textbf{MS}).
				\item \textbf{Large heap:} GC runs rarely, fast program wins (\textbf{SS}).
				\item \textbf{MC:} \alert{collecting is too slow}.
			\end{itemize}

			\medskip
			The crossing lines are the classic \textbf{space-time trade-off}.

\section{Mark-Region Collection}

\begin{frame}{Mark-Region Collectors}
	\begin{itemize}
		\item Memory is divided into \textbf{fixed-size regions}. Objects can't span regions.
		\item \textbf{Bump allocation:} each new object goes right after the previous one, by advancing (\emph{bumping}) a pointer. When no free region is left, we mark and sweep again.
		\item A region holding \alert{any} live object is \emph{unavailable}, all others are \emph{free}.
	\end{itemize}

\begin{frame}{Exact Line Marking}
	\begin{itemize}
		\item Every line a live object lies on must be marked, or its space could be reused.
		\item \textbf{Exact marking}, for each live object:
		      \begin{enumerate}
			      \item read its type to find its \textbf{size},
			      \item loop over its memory, marking \textbf{each line}.
		      \end{enumerate}
		\item \textbf{Mark-sweep} just sets \textbf{one bit} in the header compared to line marking which is \alert{expensive}.
	\end{itemize}

	\begin{exampleblock}{Key observation}
		\vspace{0.1em}
		Most objects are smaller than a line (128B), so they span \textbf{at most two lines}.
	\end{exampleblock}
\end{frame}

\begin{frame}{Conservative Line Marking}
	\begin{itemize}
		\item \textbf{Small object} (up to a line): mark only its \textbf{start line}. No size lookup, one write.
		\item Its possible second line: the allocator \textbf{skips the first line} of every hole.
		\item \textbf{Medium object} (larger than a line): marked exactly, but rare.
		\item One \textbf{header bit}, set at allocation, says small or medium.
	\end{itemize}

\section{Conclusion}

\begin{frame}{Related Work}
	\textbf{Mark-region collectors existed}, in JRockit and IBM's JVM, but were never described or evaluated in a publication.

	\medskip
	How immix differs:

	\begin{itemize}
		\item \textbf{Defragmentation:} JRockit and IBM use \alert{compaction}; immix evacuates opportunistically in a \textbf{single pass}.
		\item \textbf{Granularity:} IBM never reuses gaps smaller than 512 bytes; immix reclaims \textbf{128-byte lines}.
		\item \textbf{Metadata:} 0.8\% of the heap for immix vs 3.1\% for IBM.
	\end{itemize}

	\begin{exampleblock}{Closest relative}
		\vspace{0.1em}
		Metronome also defragments on demand, but needs exact fragmentation statistics from the previous collection.
	\end{exampleblock}
\end{frame}

\begin{frame}{Conclusion}
	\placeholder{Mark-region is a new collector family; opportunistic defragmentation mixes copying and marking in one pass; immix achieves all three performance objectives.}
\end{frame}
