\section{Garbage Collection Background}

    \placeholder{Managed languages (Java, C\#) allocate many small, short-to-medium lived objects, so the collector directly determines program performance.}

\section{Mark-Region Collection}

\begin{frame}{Mark-Region Collectors}
    \placeholder{Sweep-to-region: memory is split into regions, a region with any live object is unavailable, the rest are bump allocated into.}

    \diagramplaceholder{naive mark-region heap: fixed regions, each either free or unavailable because it holds a live object}
\end{frame}

\begin{frame}{Two Open Questions}
    \placeholder{(1) How big should regions be? Large regions waste space, small ones slow collection. (2) How to defragment, since one live object pins a whole region?}
\end{frame}

\begin{frame}{Blocks and Lines}
    \placeholder{Immix manages memory at two granularities: coarse 32KB blocks and fine 128B lines. Objects may span lines but not blocks.}

    \diagramplaceholder{heap $\to$ 32KB blocks $\to$ 256 lines of 128B each, with small objects inside lines}
\end{frame}

\begin{frame}{Initial Allocation}
    \placeholder{All blocks start empty. A thread-local allocator takes an empty block from the global allocator and bump allocates into it until the heap is exhausted, which triggers a collection.}

    \diagramplaceholder{global free block pool handing empty blocks to a thread-local allocator, which fills each block with a bump pointer}
\end{frame}

\begin{frame}{Identification and Reclamation}
    \placeholder{Tracing marks objects and their lines in a line map. A coarse-grained sweep classifies blocks as free, recyclable, or unavailable.}

    \diagramplaceholder{line map of several blocks after marking, each block labelled free / recyclable / unavailable}
\end{frame}

\begin{frame}{Steady State Allocation}
    \placeholder{Threads allocate into recycled blocks in address order: scan the line map for a hole, bump allocate into it, move to the next hole. Empty blocks are used only after all recyclable blocks.}

    \diagramplaceholder{recycled block with marked and free lines, cursor and limit pointing to the current hole, then jumping to the next hole}
\end{frame}

\begin{frame}{Immix Heap Organization}
    \placeholder{Walk through the paper's figure: global free block allocator, recycled blocks, cursor and limit, line mark legend.}

    \paperplaceholder{Figure 2 (immix heap organization)}
\end{frame}

\begin{frame}{Recycling and Allocation Policies}
    \placeholder{A block with at least $F = 1$ free line is recyclable. Recyclable blocks are used in address order, free blocks last because the large object space also needs them.}
\end{frame}

\begin{frame}{Parallelism}
    \placeholder{Synchronized global block allocator, unsynchronized thread-local allocators. Racy marking is safe; line marks are bytes to avoid synchronization.}
\end{frame}

\begin{frame}{Demand-Driven Overflow Allocation}
    \placeholder{Medium objects (larger than a line) that do not fit the current hole go to a separate overflow allocator using empty blocks, so small holes are not skipped.}

    \diagramplaceholder{medium object too large for the current hole, redirected to an empty overflow block instead of skipping the hole}
\end{frame}

\section{Opportunistic Defragmentation}

\begin{frame}{Why Defragment?}
    \placeholder{A pure mark-region collector never moves objects, so fragmented blocks stay partially used forever.}
\end{frame}

\begin{frame}{Opportunistic Evacuation}
    \placeholder{Before collection, choose candidate blocks to evacuate and target space. During marking, move objects off candidates and leave forwarding pointers; stop moving when space runs out. 2.5\% of the heap is reserved as evacuation headroom.}

    \diagramplaceholder{fragmented candidate block, live objects copied into holes of a target block, forwarding pointers left behind}
\end{frame}

\begin{frame}{Candidate Selection}
    \placeholder{Defragment on demand. Mark and available histograms indexed by hole count; select the most fragmented blocks that fit the available space.}

    \diagramplaceholder{mark and available histograms by hole count, with a cut-off line where required space exceeds available space}
\end{frame}

\begin{frame}{Mixing Marking and Evacuation}
    \placeholder{Per-object decision during the trace: unmarked and unmovable $\to$ mark; unmarked and movable $\to$ evacuate and forward; forwarded $\to$ update reference. Single pass.}

    \diagramplaceholder{flowchart of the per-object decision during tracing}
\end{frame}

\begin{frame}{Parallel Evacuation and Pinning}
    \placeholder{Compare-and-swap to race on forwarding, paid only in the specialized defragmenting trace. Pinned objects are simply marked in place.}
\end{frame}

\section{Implementation Details}

\begin{frame}{Block and Line Size}
    \placeholder{128B lines roughly match cache lines and fit 1--4 objects; 32KB blocks roughly match pages. Worst-case fragmentation bounds: about 25\% per block, 97\% per line in theory. Metadata is one byte per line and four per block: 0.8\% overhead.}
\end{frame}

\begin{frame}{Conservative Line Marking}
    \placeholder{Most objects are under 128B and span at most two lines. Mark only the start line; the allocator skips the first line of each hole, which implicitly covers the second line. Medium objects are marked exactly.}

    \diagramplaceholder{small object straddling two lines: only the first line is marked, the next line is treated as implicitly marked}
\end{frame}

\begin{frame}{Optimizing Hot Paths}
    \placeholder{The allocator matches semi-space speed and tracing comes close to mark-sweep. Lines are marked when an object is scanned, not when it is marked, which hides the cost.}
\end{frame}
